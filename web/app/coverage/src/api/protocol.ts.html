export function blobHashToU64(hash: number[] | number): number {
  if (typeof hash === "number") return hash;
  if (Array.isArray(hash) && hash.length >= 8) {
    let result = 0;
    for (let i = 0; i < 8; i++) {
      result = result * 256 + (hash[i] || 0);
    }
    return result;
  }
  return 0;
}
